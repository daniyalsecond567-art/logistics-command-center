// Trucking-1 Firestore Security Rules — COMPLETE
// Deploy in Firebase Console → Firestore Database → Rules
//
// Collection summary
// ------------------
//  trainingAudio  — Public READ (anyone can browse the training briefs).
//                   Admin WRITE (create/update/delete require an authenticated
//                   Firebase user; the app has no public sign-up flow, so any
//                   successful login is an administrator).
//
//  userRequests   — Public CREATE with strict validation (short text, must
//                   start as "Pending"). Admin READ + DELETE only.
//                   No updates allowed (requests are immutable).
//
//  Everything else — Default denied. Add explicit blocks below if a new
//                    collection is introduced.
//
// Notes
// -----
//  • Firebase Authentication itself (Email/Password sign-in) does NOT need
//    Firestore rules — it is handled by the Auth service.
//  • Cloudinary and Supabase Storage are external services; their access is
//    controlled by their own dashboards (unsigned upload preset for
//    Cloudinary, bucket policies for Supabase). Firestore rules do not
//    apply to them.
//  • Audio files themselves live on Cloudinary / Supabase; Firestore only
//    stores the resulting metadata (title, note, single audioUrl, duration,
//    displayOrder, createdAt).

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ---------------------------------------------------------------
    // Training audio metadata
    // Public read · Admin (any authenticated user) write
    // ---------------------------------------------------------------
    match /trainingAudio/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;
    }

    // ---------------------------------------------------------------
    // User requests submitted from the Training page
    // Public create (validated) · Admin read + delete · No updates
    // ---------------------------------------------------------------
    match /userRequests/{docId} {
      // Anyone may submit a short request. Enforce shape + length + status.
      allow create: if
        request.resource.data.keys().hasOnly(['text', 'status', 'createdAt']) &&
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() <= 20 &&
        request.resource.data.status == 'Pending';

      // Only administrators (authenticated users) can view, update, or remove requests.
      // Updates are used to record Approved / Declined decisions with a timestamp.
      allow read, update, delete: if request.auth != null;
    }

    // Issue reports from the Training page. Public create, admin read/delete.
    match /userIssues/{docId} {
      allow create: if
        request.resource.data.keys().hasOnly(['text', 'status', 'createdAt']) &&
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() <= 300 &&
        request.resource.data.status == 'Open';
      allow read, update, delete: if request.auth != null;
    }

    // ---------------------------------------------------------------
    // Site configuration (maintenance mode toggle, etc.)
    // Public read (so the app can gate the UI) · Admin write only
    // ---------------------------------------------------------------
    match /siteConfig/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;
    }

    // ---------------------------------------------------------------
    // Default deny — every other path is closed until an explicit
    // block above allows it.
    // ---------------------------------------------------------------
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
