// Trucking-1 Firestore Security Rules
// Deploy in Firebase Console → Firestore Database → Rules

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Training audio metadata
    match /trainingAudio/{docId} {
      // Anyone (including unauthenticated visitors) can read training briefs.
      allow read: if true;

      // Only authenticated administrators can create, edit, or delete.
      // Admin accounts are provisioned in the Firebase Console — the app has
      // no public sign-up flow, so any successful login = admin.
      allow create, update, delete: if request.auth != null;
    }

    // Default deny for anything else.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
