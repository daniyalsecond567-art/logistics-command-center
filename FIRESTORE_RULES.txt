// Trucking-1 Firestore Security Rules
// Deploy in Firebase Console → Firestore Database → Rules

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Training audio metadata
    match /trainingAudio/{docId} {
      // Anyone (including unauthenticated visitors) can read training briefs.
      allow read: if true;

      // Only authenticated administrators can create, edit, or delete.
      // Admin accounts are provisioned in the Firebase Console — the app has
      // no public sign-up flow, so any successful login = admin.
      allow create, update, delete: if request.auth != null;
    }

    // User requests submitted from the Training page.
    match /userRequests/{docId} {
      // Anyone can submit a request; keep it short and pending.
      allow create: if
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() <= 20 &&
        request.resource.data.status == "Pending";

      // Only admins can read or delete requests.
      allow read, delete: if request.auth != null;

      // No editing.
      allow update: if false;
    }


    // Default deny for anything else.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
