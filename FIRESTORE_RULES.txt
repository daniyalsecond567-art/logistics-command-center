// Trucking-1 Firestore Security Rules — COMPLETE
// Deploy in Firebase Console → Firestore Database → Rules
//
// Collection summary
// ------------------
//  trainingAudio  — Public READ (anyone can browse the training briefs).
//                   Admin WRITE (create/update/delete require an authenticated
//                   Firebase user; the app has no public sign-up flow, so any
//                   successful login is an administrator).
//
//  userRequests   — Public CREATE with strict validation (short text, must
//                   start as "Pending"). Admin READ + DELETE only.
//                   No updates allowed (requests are immutable).
//
//  Everything else — Default denied. Add explicit blocks below if a new
//                    collection is introduced.
//
// Notes
// -----
//  • Firebase Authentication itself (Email/Password sign-in) does NOT need
//    Firestore rules — it is handled by the Auth service.
//  • Cloudinary and Supabase Storage are external services; their access is
//    controlled by their own dashboards (unsigned upload preset for
//    Cloudinary, bucket policies for Supabase). Firestore rules do not
//    apply to them.
//  • Audio files themselves live on Cloudinary / Supabase; Firestore only
//    stores the resulting metadata (title, note, single audioUrl, duration,
//    displayOrder, createdAt).

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ---------------------------------------------------------------
    // Training audio metadata
    // Public read · Admin (any authenticated user) write
    // ---------------------------------------------------------------
    match /trainingAudio/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;
    }

    // ---------------------------------------------------------------
    // User requests submitted from the Training page
    // Public create (validated) · Admin read + delete · No updates
    // ---------------------------------------------------------------
    match /userRequests/{docId} {
      // Anyone may submit a short request. Enforce shape + length + status.
      allow create: if
        request.resource.data.keys().hasOnly(['text', 'status', 'createdAt']) &&
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() <= 20 &&
        request.resource.data.status == 'Pending';

      // Only administrators (authenticated users) can view, update, or remove requests.
      // Updates are used to record Approved / Declined decisions with a timestamp.
      allow read, update, delete: if request.auth != null;
    }

    // Issue reports from the Training page. Public create, admin read/delete.
    match /userIssues/{docId} {
      allow create: if
        request.resource.data.keys().hasOnly(['text', 'status', 'createdAt']) &&
        request.resource.data.text is string &&
        request.resource.data.text.size() > 0 &&
        request.resource.data.text.size() <= 300 &&
        request.resource.data.status == 'Open';
      allow read, update, delete: if request.auth != null;
    }

    // Training favourites — each signed-in user can only touch their own doc.
    match /userFavorites/{uid} {
      allow read, delete: if request.auth != null && request.auth.uid == uid;
      allow create, update: if request.auth != null && request.auth.uid == uid &&
        request.resource.data.keys().hasOnly(['ids', 'updatedAt']) &&
        request.resource.data.ids is list &&
        request.resource.data.ids.size() <= 1000;
    }

    // Songs added from the Training page (YouTube embeds). Public read,
    // public validated create (doc id = yt_<videoId> blocks duplicates,
    // since existing docs can't be overwritten). Admin delete only.
    match /trainingSongs/{docId} {
      allow read: if true;
      allow create: if
        request.resource.data.keys().hasOnly(['title', 'sourceUrl', 'videoId', 'type', 'creatorId', 'createdAt']) &&
        request.resource.data.title is string &&
        request.resource.data.title.size() > 0 &&
        request.resource.data.title.size() <= 60 &&
        request.resource.data.type == 'youtube' &&
        request.resource.data.videoId is string &&
        request.resource.data.videoId.matches('^[A-Za-z0-9_-]{11}$') &&
        docId == 'yt_' + request.resource.data.videoId &&
        request.resource.data.sourceUrl == 'https://www.youtube.com/watch?v=' + request.resource.data.videoId &&
        request.resource.data.creatorId is string &&
        request.resource.data.creatorId.size() <= 128 &&
        request.resource.data.createdAt == request.time;
      allow update, delete: if request.auth != null;
    }

    // ---------------------------------------------------------------
    // Site configuration (maintenance mode toggle, etc.)
    // Public read (so the app can gate the UI) · Admin write only
    // ---------------------------------------------------------------
    match /siteConfig/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null;
    }

    // Admin change history. Anyone may append a validated entry
    // (song additions, favourites, requests come from visitors).
    // Only admins can read or delete. Entries are never edited.
    match /activityLog/{docId} {
      allow create: if
        request.resource.data.keys().hasOnly(['action', 'detail', 'actor', 'device', 'createdAt']) &&
        request.resource.data.action is string && request.resource.data.action.size() > 0 && request.resource.data.action.size() <= 40 &&
        request.resource.data.detail is string && request.resource.data.detail.size() <= 200 &&
        request.resource.data.actor is string && request.resource.data.actor.size() <= 100 &&
        request.resource.data.device is string && request.resource.data.device.size() <= 80 &&
        request.resource.data.createdAt == request.time;
      allow read, delete: if request.auth != null;
      allow update: if false;
    }

    // ---------------------------------------------------------------
    // Default deny — every other path is closed until an explicit
    // block above allows it.
    // ---------------------------------------------------------------
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
